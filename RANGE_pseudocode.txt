// DATA PREPROCESSING
BEGIN
	// Read Raw UWB Dataset
	READ datasets
    
    // Filter for LOS Signals only
    REMOVE NLOS entries
    
    // Feature Selection
    REMOVE constant features = ["CH", "BITRATE", "PRFR"]
    REMOVE correlated features:
    	DROP STDEV_NOISE (correlated with MAX_NOISE)
        DROP FP_AMP3 (correlated with FP_AMP2)
        
    // CIR Processing
    NORMALIZE CIR0-CIR1015 using RXPACC
    RETAIN CIR taps with |r| > 0.2 (correlation with RANGE)
    APPLY PCA:
    	KEEP top 6 compontents (CIR_PC1 to CIR_PC6)
       
    // Merge & Finalise
    MERGE PCA Components with Core Features
    VALIDATE features:
    	CALCULATE Random Forest importance
        CALCULATE Paired t-tests on feature removal
    FINALIZE dataset with 14 features
END

// DATA MINING
BEGIN
	// Read Finalized dataset (14 features)
    READ dataset
    
    SPLIT dataset into training and testing (70:30)
    TRAIN linear regression model on training set
    PREDICT "RANGE" using model on the testing set

END

// DATA POSTPROCESSING
BEGIN
	// Evaluate predictions
    CALCULATE Mean Absolute Error ON predictions
    CALCULATE Mean Squared Error ON predictions
    CALCULATE Root Mean Squared Error ON predictions
    CALCULATE R2 Score ON predictions
    
    // Visulize results
    PLOT Actual vs Predicted RANGE (Scatter Plot)
    PLOT Feature importance (based on model coefficients)
    PLOT Permutation importance (Bar Plot)
END
